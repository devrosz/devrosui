export const onErrorCode = 
`
"use client"

import { InputOTP } from "@devrosui/react"
import { useState, useId } from "react"

export default function MyComponent() {
    
    const [error, setError ] = useState(null)
    const errorKey = useId()

    async function handleSubmit(input) {
        try {
            // someAPI(input)
        } catch(e) {
            setError({
                message: e.message,
                key: errorKey
            })
        }
    }

  return (
        <InputOTP>
            <InputOTP.Form
                onSubmit={handleSubmit}
                errorMessage={error}
                inputLength={6}
            >
                <InputOTP.Slot index={0} />
                <InputOTP.Slot index={1} />
                <InputOTP.Slot index={2} />
                <InputOTP.Separator />
                <InputOTP.Slot index={3} />
                <InputOTP.Slot index={4} />
                <InputOTP.Slot index={5} />
            </InputOTP.Form>
        </InputOTP>
    )
}

`

    <h4 id="error">On Error</h4>

    It is possible to pass custom errors to ```InputOTP``` like network errors, API-errors etc.
    The ```key``` value is mandatory and must be different for each time an error occurs in order
    to trigger a re-render which will display the error again. The ```message``` value can be the same.

    <h4 id="props-form">InputOTP.Form</h4>
    
    <div className="docs-page-table-container">
        | Prop | Type | Default | Description |
        | :--- | :--- | :------ | :---------- |
        | ```inputLength``` | ```number``` | - | Amount of slots |
        | ```errorMessage``` | ```{message: string, key: string}``` | - | Custom error message |
        | ```onSubmit``` | ```(string) => Promise<void> \| void``` | - | Callback function that fires on completion |
        | ```autoSubmit``` | ```boolean``` | true | Automatically calls ```onSubmit``` on completion |
        | ```disabled``` | ```boolean``` | false | Prohibits user from filling in the input |
        | ```allowNumbers``` | ```boolean``` | true | Allow input value to contain numbers |
        | ```allowLetters``` | ```boolean``` | true | Allow input value to contain letters |
        | ```children``` | ```ReactNode``` | - | Expecting ```<InputOTP.Slot>``` and in if applicable also ```<InputOTP.Separator>``` |
    </div>
