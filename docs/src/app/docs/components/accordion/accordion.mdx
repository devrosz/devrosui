export function getAccordion(variant) {
    const isFilled = !variant.toLowerCase().includes("empty")
    const hasChevron = variant.toLowerCase().includes("chevron")

    return (
        <div
            className="accordion-container"
            style={{margin: "0 auto", width: "60%"}}
        >
            <Accordion
                items={accordionItems}
                background={isFilled ? "filled" : "empty"}
                toggleIcon={hasChevron ? "chevron" : "plus"}
            />
        </div>
    )
}
