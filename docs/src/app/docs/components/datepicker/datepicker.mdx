export const usageCode =
`
"use client"

import { DatePicker } from "@devrosui/react"
import React from "react"

export default function MyComponent() {

    const [formDate, setFormData] = React.useState({
        date: "",
        ... // add more fields according to your form.
    })

    function handleChange(name, value) {
        setFormData(prev => {
            return {
                ...prev,
                [name]: value
            }
        })
    }

    return (
        <DatePicker 
            id="datepicker" 
            name="date"
            date={formDate.date}
            onChange={handleChange}
        />
    )
}
`
