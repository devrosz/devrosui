export const usageCode =
`
"use client"

import { Select } from "@devrosui/react"
import React from "react"

export default function MyComponent() {

    const [formDate, setFormData] = React.useState({
        carBrand: "",
        ... // add more fields according to your form.
    })

    function handleChange(name, value) {
        setFormData(prev => {
            return {
                ...prev,
                [name]: value
            }
        })
    }

    return (
        <Select 
            id="car-brand" 
            name="carBrand"
            values={formDate.carBrand}
            onSelect={handleChange}
            options={[
                "Volkswagen",
                "Audi",
                "Mercedes",
                "BMW"
            ]}
            placeholder="Select car brand"
            label="Car brand"
        />
    )
}
`
