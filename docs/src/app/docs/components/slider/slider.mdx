export const connectcode =
`
"use client"

import { Slider } from "@devrosui/react"
import React from "react"

export default function MyComponent() {

    const [value, setValue] = React.useState("0")

    return (
        <Slider 
            id="slider" 
            name="slider"
            value={value}
            setValue={setValue}
            min="0"
            max="100"
            label="Range"
        />
    )
}
`

    <h3 id="props">Props</h3>

    <div className="docs-page-table-container">
        | Prop | Type | Default | Description |
        | :--- | :--- | :------ | :---------- |
        | ```id*``` | ```string``` | - | id of slider input |
        | ```name*``` | ```string``` | - | name of slider input |
        | ```min*``` | ```number``` | ```0``` | minimum value of the input |
        | ```max*``` | ```number``` | ```100``` | maximum value of the input |
        | ```value``` | ```number``` | ```min``` | current value of the input |
        | ```setValue``` | ```(number) => void``` | - | callback function that updates the current value in the parent |
        | ```label``` | ```string``` | - | Label above the range input |
        | ```description``` | ```string```| ```""``` | displays a description about the input |
        | ```disabled``` | ```boolean``` | ```false``` | disables/enables range input  |
        | ```required``` | ```boolean``` | false | Forces user to fill in the input |
        | ```unit``` | ```string``` | - | unit of the value in the range input |
        | ```width``` | ```string```| ```15rem``` | sets the width of the input |
        | ```step``` | ```string```| ```1``` | step size of the slider |
        | ```showValue``` | ```boolean```| ```true``` | Toggles showing the value |
    </div>
